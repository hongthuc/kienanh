    function handleDriveImg(imgEl, fileId, fallbackLabel) {
      const stage = imgEl.getAttribute('data-fallback-stage') || '0';
      if (stage === '0') {
        imgEl.setAttribute('data-fallback-stage', '1');
        imgEl.src = 'https://lh3.googleusercontent.com/d/' + fileId + '=w1000';
      } else if (stage === '1') {
        imgEl.setAttribute('data-fallback-stage', '2');
        imgEl.src = 'https://drive.google.com/uc?export=view&id=' + fileId;
      } else {
        imgEl.onerror = null;
        imgEl.src = 'https://placehold.co/600x600/f8fafc/b91c1c?text=' + encodeURIComponent(fallbackLabel || 'Kien+Anh+Bakery');
      }
    }

      slides.forEach((slide, idx) => {

        }
      });

    // 5. Lọc danh mục bài viết trang chủ
    function filterPosts(cat, btnEl) {
      const cards = document.querySelectorAll('.blog-card');
      const btns = document.querySelectorAll('.post-cat-btn');
      btns.forEach(b => {
        b.className = 'post-cat-btn cat-nav-item px-2.5 py-1 rounded bg-slate-200 text-slate-700 hover:bg-slate-300';
      });
      if (btnEl) {
        btnEl.className = 'post-cat-btn cat-nav-item px-2.5 py-1 rounded bg-brand text-white';
      }
      cards.forEach(card => {
        const cardCat = card.getAttribute('data-post-cat');
        if (cat === 'all' || cardCat === cat) {
          card.classList.remove('hidden');
        } else {
          card.classList.add('hidden');
        }
      });
    }
